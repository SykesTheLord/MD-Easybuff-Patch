# +Easybuff - MD Systems
# Thin wrappers over Millennium Dawn's own scripted effects. Nothing here reimplements MD
# logic: every wrapper converts a "set to value" cheat into the "change by delta" call MD
# already exposes, so MD's clamps, tooltips and refresh hooks all still run.
#
# Where a name below differs from MD's published scripted-effects reference
# (millenniumdawn.github.io), the published page documents the released 1.12.x mod and this
# patch targets 2.0. The 2.0 code name is used and the difference is noted at the call site.

# Economy
# ---------------------------------------------------------------------------

# Required input: ebmd_target - the debt value to end up at
ebmd_set_debt = {
	set_temp_variable = { debt_change = ebmd_target }
	subtract_from_temp_variable = { debt_change = debt }
	modify_debt_effect = yes
	ingame_update_setup = yes
}

# Required input: ebmd_ratio - debt as a multiple of GDP. Used to push interest_rate across
# the 15% and 25% thresholds through MD's real formula (debt / gdp_total * 10) rather than
# by setting interest_rate, which calculate_interest_rate would overwrite without ever
# applying the threshold penalties.
ebmd_set_debt_gdp_ratio = {
	set_temp_variable = { ebmd_target = gdp_total }
	multiply_temp_variable = { ebmd_target = ebmd_ratio }
	ebmd_set_debt = yes
}

# Required input: ebmd_target - the treasury value to end up at.
# Clamped at 0: MD's automated_debt_taker turns a negative treasury into new debt plus a 1%
# fee on the next weekly tick, which would silently undo the cheat and raise interest.
ebmd_set_treasury = {
	clamp_temp_variable = { var = ebmd_target min = 0 }
	set_temp_variable = { treasury_change = ebmd_target }
	subtract_from_temp_variable = { treasury_change = treasury }
	modify_treasury_effect = yes
}

# Required input: ebmd_target - the corporate tax rate to end up at. MD clamps 0-50.
ebmd_set_corporate_tax = {
	enable_corporate_tax_rate_change = yes
	set_temp_variable = { corp_change = ebmd_target }
	subtract_from_temp_variable = { corp_change = corporate_tax_rate }
	modify_corporate_tax_rate_effect = yes
}

# Required input: ebmd_factor - multiplier applied to the current GDP.
# Transient by design: calculate_gdp resets gdp_total to 0 and re-sums it from the
# gdp_from_* sector terms on the next ingame_update_setup, then re-derives gdp_per_capita
# from it. The caller fires ebmd.11 to explain this.
ebmd_scale_gdp = {
	multiply_variable = { gdp_total = ebmd_factor }
	set_variable = {
		gdp_per_capita = {
			value = gdp_total
			divide = population_total_m
			clamp = { min = 0 max = 100000 }
		}
	}
	update_money_dirty_variable = yes
}

# MD's Expand Money Supply button grants monetary_expand_active_idea (seigniorage income
# +25%) but charges currency_strength -0.05 plus a 545-730 day cooldown flag. This grants
# the idea without either price.
# Required input: ebmd_days - 0 for an untimed idea.
ebmd_grant_monetary_expansion = {
	if = {
		limit = { check_variable = { ebmd_days > 0 } }
		add_timed_idea = { idea = monetary_expand_active_idea days = ebmd_days }
	}
	else = {
		add_ideas = monetary_expand_active_idea
	}
	ingame_update_setup = yes
}

ebmd_clear_monetary_cooldown = {
	clr_country_flag = monetary_expand_cooldown
	clr_country_flag = monetary_austerity_cooldown
	clr_country_flag = printed_money_recently
}

# MD's Print Money button adds treasury but costs 50 PP, sets printed_money_recently and
# accumulates print_money_base_add, which multiplies next quarter's inflation. This adds the
# money and touches none of that.
# Required input: ebmd_target - amount to add.
ebmd_print_money_free = {
	set_temp_variable = { treasury_change = ebmd_target }
	modify_treasury_effect = yes
}

# Required input: ebmd_target - the inflation rate to end up at.
# inflation_rate_var only recomputes quarterly (quarterly_inflation_calculation, from
# calculate_gdp on months 3/6/9/12), so this holds for up to three months. The follow-up
# call is required: without it the seven derived inflation variables keep their old values
# and the cheat appears to do nothing.
ebmd_set_inflation = {
	set_variable = { inflation_rate_var = ebmd_target }
	set_variable = { print_money_base_add = 0 }
	calculate_inflation_effects = yes
	ingame_update_setup = yes
}

# Required input: ebmd_target - the central bank policy rate to end up at. MD clamps 0-20.
# Raising it lowers inflation next quarter but raises interest_rate immediately, because
# cb_rate_premium = cb_policy_rate * 0.5 feeds interest_rate.
ebmd_set_cb_rate = {
	set_variable = { cb_policy_rate = ebmd_target }
	clamp_variable = { var = cb_policy_rate min = 0 max = 20 }
	ingame_update_setup = yes
}

# remove_org fixes up eleven parallel global arrays plus every enrolled country's intel
# arrays, so slots are always torn down from index 0 rather than edited directly.
# The counter is a backstop: if a future MD change ever made remove_org a no-op here, the
# array length would stop falling and the loop would hang the game.
ebmd_clear_terror_orgs = {
	set_temp_variable = { ebmd_guard = 0 }
	while_loop_effect = {
		limit = {
			check_variable = { global.active_terror_orgs^num > 0 }
			check_variable = { ebmd_guard < 64 }
		}
		set_temp_variable = { targeted_org = 0 }
		remove_org = yes
		add_to_temp_variable = { ebmd_guard = 1 }
	}
}

# Bumping a variable the scripted GUI declares as `dirty` is what makes the readout
# redraw. Wrapping it keeps every button from having to remember.
ebmd_gui_refresh = {
	add_to_variable = { ebmd_gui_dirty = 1 }
	if = {
		limit = { check_variable = { ebmd_gui_dirty > 1000000 } }
		set_variable = { ebmd_gui_dirty = 0 }
	}
}

# Microchip plants
# ---------------------------------------------------------------------------

# One microchip plant in every owned, controlled state, in the style of Easybuff's
# per-state infrastructure / air base options - plus one extra shared building slot per
# state, so the plant never eats a slot the player already had.
# The slot is added first: add_building_construction needs a free shared slot to land.
# States at MD's cap (state_max = 5) are skipped, or they would gain a slot and no plant.
# Owned as well as controlled, matching MD's free_shared_building_slots: a plant built in
# occupied territory is lost at the peace deal.
ebmd_microchip_plant_every_state = {
	every_controlled_state = {
		limit = {
			is_owned_by = ROOT
			microchip_plant < 5
		}
		add_extra_state_shared_building_slots = 1
		add_building_construction = {
			type = microchip_plant
			level = 1
			instant_build = yes
		}
	}
	# MD sums microchip_plant_total per state on ingame_update_setup; refresh so the
	# resource, tax and energy figures include the new plants now instead of next week.
	ingame_update_setup = yes
	ebmd_refresh_energy = yes
}

# Required input: ebmd_pop_delta_m - population to add (negative removes), in millions.
# Split across the states ebmd_staff_factories weighted, in proportion to ebmd_staff_weight.
# add_manpower is chunked at 1,000,000 people: HOI4 variables overflow a little above
# 2 million, and MD itself caps script maths at 1,000,000 (@math_expressions_max).
# In state scope add_manpower changes the state's population, which is what MD sums
# into population_total_m.
ebmd_apply_population_delta = {
	every_owned_state = {
		limit = {
			is_controlled_by = ROOT
			check_variable = { ebmd_staff_weight > 0 }
		}
		set_temp_variable = { ebmd_share_m = ebmd_pop_delta_m }
		multiply_temp_variable = { ebmd_share_m = ebmd_staff_weight }
		divide_temp_variable = { ebmd_share_m = ebmd_staff_weight_total }
		set_temp_variable = { ebmd_chunk_guard = 0 }
		while_loop_effect = {
			limit = {
				check_variable = { ebmd_share_m > 1 }
				check_variable = { ebmd_chunk_guard < 5000 }
			}
			add_manpower = 1000000
			subtract_from_temp_variable = { ebmd_share_m = 1 }
			add_to_temp_variable = { ebmd_chunk_guard = 1 }
		}
		while_loop_effect = {
			limit = {
				check_variable = { ebmd_share_m < -1 }
				check_variable = { ebmd_chunk_guard < 5000 }
			}
			add_manpower = -1000000
			add_to_temp_variable = { ebmd_share_m = 1 }
			add_to_temp_variable = { ebmd_chunk_guard = 1 }
		}
		multiply_temp_variable = { ebmd_share_m = 1000000 }
		add_manpower = ebmd_share_m
	}
	add_to_temp_variable = { ebmd_pop_added_m = ebmd_pop_delta_m }
}

# Adds population to factory states until no sector is short of workers.
#
# MD pools workers nationally (calculate_gdp): 60% of population is the workforce, and each
# sector hires in a fixed order, capped by what is left. total_unemployed is the leftover
# after the last sector, so unemployment near zero means a sector ran out of workers.
# Several sectors scale with population themselves (subsistence agriculture, healthcare,
# bureaucracy, the resource-sector cap), so the shortfall is not computed in closed form -
# MD's own ingame_update_setup is re-run after each step and its unemployment is the oracle.
#
# Grow in doubling steps until unemployment reaches 2%, then, if that overshot past 5%,
# bisect back into 2-5%. 6% is where MD's high-unemployment penalties begin.
ebmd_staff_factories = {
	# Weight each owned, controlled state by the buildings that employ MD's workforce, so
	# the people land where the jobs are.
	set_temp_variable = { ebmd_staff_weight_total = 0 }
	every_owned_state = {
		limit = { is_controlled_by = ROOT }
		set_variable = {
			ebmd_staff_weight = {
				value = building_level@industrial_complex
				add = building_level@arms_factory
				add = building_level@dockyard
				add = building_level@offices
				add = building_level@microchip_plant
				add = building_level@composite_plant
			}
		}
		add_to_temp_variable = { ebmd_staff_weight_total = ebmd_staff_weight }
	}
	if = {
		limit = { check_variable = { ebmd_staff_weight_total = 0 } }
		capital_scope = { set_variable = { ebmd_staff_weight = 1 } }
		set_temp_variable = { ebmd_staff_weight_total = 1 }
	}

	ingame_update_setup = yes
	set_temp_variable = { ebmd_pop_added_m = 0 }
	set_temp_variable = { ebmd_pop_lo_m = 0 }
	set_temp_variable = { ebmd_staff_failed = 0 }
	set_temp_variable = { ebmd_pop_step_m = population_total_m }
	multiply_temp_variable = { ebmd_pop_step_m = 0.05 }
	clamp_temp_variable = { var = ebmd_pop_step_m min = 0.01 }

	# Grow.
	set_temp_variable = { ebmd_guard = 0 }
	while_loop_effect = {
		limit = {
			check_variable = { total_unemployed_percentage_display < 0.02 }
			check_variable = { ebmd_guard < 16 }
		}
		set_temp_variable = { ebmd_pop_before_m = population_total_m }
		set_temp_variable = { ebmd_pop_lo_m = ebmd_pop_added_m }
		set_temp_variable = { ebmd_pop_delta_m = ebmd_pop_step_m }
		ebmd_apply_population_delta = yes
		ingame_update_setup = yes
		# If population did not move, the loop would add people forever without effect.
		if = {
			limit = { NOT = { check_variable = { population_total_m > ebmd_pop_before_m } } }
			set_temp_variable = { ebmd_staff_failed = 1 }
			set_temp_variable = { ebmd_guard = 99 }
		}
		multiply_temp_variable = { ebmd_pop_step_m = 2 }
		add_to_temp_variable = { ebmd_guard = 1 }
	}

	# Trim an overshoot by bisecting between the last short amount (lo) and the first
	# sufficient one (hi).
	if = {
		limit = {
			check_variable = { ebmd_staff_failed = 0 }
			check_variable = { total_unemployed_percentage_display > 0.05 }
		}
		set_temp_variable = { ebmd_pop_hi_m = ebmd_pop_added_m }
		set_temp_variable = { ebmd_guard = 0 }
		while_loop_effect = {
			limit = {
				OR = {
					check_variable = { total_unemployed_percentage_display < 0.02 }
					check_variable = { total_unemployed_percentage_display > 0.05 }
				}
				check_variable = { ebmd_guard < 10 }
			}
			set_temp_variable = { ebmd_pop_mid_m = ebmd_pop_lo_m }
			add_to_temp_variable = { ebmd_pop_mid_m = ebmd_pop_hi_m }
			multiply_temp_variable = { ebmd_pop_mid_m = 0.5 }
			set_temp_variable = { ebmd_pop_delta_m = ebmd_pop_mid_m }
			subtract_from_temp_variable = { ebmd_pop_delta_m = ebmd_pop_added_m }
			ebmd_apply_population_delta = yes
			ingame_update_setup = yes
			if = {
				limit = { check_variable = { total_unemployed_percentage_display < 0.02 } }
				set_temp_variable = { ebmd_pop_lo_m = ebmd_pop_mid_m }
			}
			else = {
				set_temp_variable = { ebmd_pop_hi_m = ebmd_pop_mid_m }
			}
			add_to_temp_variable = { ebmd_guard = 1 }
		}
		# Never finish short: if bisection ran out below 2%, return to the last sufficient amount.
		if = {
			limit = { check_variable = { total_unemployed_percentage_display < 0.02 } }
			set_temp_variable = { ebmd_pop_delta_m = ebmd_pop_hi_m }
			subtract_from_temp_variable = { ebmd_pop_delta_m = ebmd_pop_added_m }
			ebmd_apply_population_delta = yes
			ingame_update_setup = yes
		}
	}

	every_owned_state = {
		clear_variable = ebmd_staff_weight
	}
}

# +1 Renewable Energy Infrastructure in every owned, controlled state. The building has
# its own slots (shares_slots = no, state_max = 20), so no shared slot is added; the
# free_building_slots check is MD's own (renewable_energy_infra_random_build_loc) and
# skips states already at the cap.
ebmd_renewables_every_state = {
	every_controlled_state = {
		limit = {
			is_owned_by = ROOT
			free_building_slots = { building = renewable_energy_infra size > 0 }
		}
		add_building_construction = {
			type = renewable_energy_infra
			level = 1
			instant_build = yes
		}
	}
	# state_renewable_energy_var is summed per state on ingame_update_setup.
	ingame_update_setup = yes
	ebmd_refresh_energy = yes
}

# One Nuclear Enrichment Facility in every owned, controlled state that lacks one
# (state_max = 1), plus what MD's own sp_nuclear_energy project does on completion
# (special_projects/projects/nuclear_projects.txt) - without it the facilities are inert:
#   nuclear_technology - enables the building; MD's one_random_enrichment_facility requires it
#   global.enrichment_countries - lists you as a reactor-material seller in MD's
#     purchase_reactor_grade_material diplomatic action; nothing else reads it
#   enabled_nuclear_reactor_fuel_production - energy_calc only produces fuel while it is set
#     (the player's own energy-view toggle clears it; this switches it back on)
#   var_reactor_material_stockpile - the project seeds 300; raised to 300, never lowered
# A shared slot is added before each build, as MD's one_random_enrichment_facility does.
ebmd_enrichment_every_state = {
	set_technology = {
		nuclear_technology = 1
		popup = no
	}
	if = {
		limit = { NOT = { is_in_array = { global.enrichment_countries = THIS.id } } }
		add_to_array = { global.enrichment_countries = THIS.id }
	}
	clamp_variable = { var = var_reactor_material_stockpile min = 300 }
	set_country_flag = enabled_nuclear_reactor_fuel_production
	every_controlled_state = {
		limit = {
			is_owned_by = ROOT
			enrichment_facility < 1
		}
		add_extra_state_shared_building_slots = 1
		add_building_construction = {
			type = enrichment_facility
			level = 1
			instant_build = yes
		}
	}
	# total_enrichment_facilities is summed per state on ingame_update_setup, and gates fuel output.
	ingame_update_setup = yes
	ebmd_refresh_energy = yes
}
